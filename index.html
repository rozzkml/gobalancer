Quickstart
Drop-in replacement for OpenAI — change one line
# Kirim request ke gateway curl https://your-project.vercel.app/v1/chat/completions \ -H "Content-Type: application/json" \ -H "Authorization: Bearer any-key" \ -d '{ "model": "gemini-3.5-flash-lite", "messages": [{"role": "user", "content": "Halo!"}] }'
# pip install openai from openai import OpenAI client = OpenAI( base_url="https://your-project.vercel.app/v1", api_key="any-key" # gateway yang handle ) response = client.chat.completions.create( model="gemini-3.5-flash-lite", # 250K TPM, agent-capable (atau: gemini-3.1-flash-lite) messages=[{"role": "user", "content": "Halo!"}] ) print(response.choices[0].message.content)
// npm install openai import OpenAI from 'openai'; const client = new OpenAI({ baseURL: 'https://your-project.vercel.app/v1', apiKey: 'any-key', dangerouslyAllowBrowser: true }); const res = await client.chat.completions.create({ model: 'gemini-3.5-flash-lite', messages: [{ role: 'user', content: 'Hello!' }] }); console.log(res.choices[0].message.content);
# Configure in Hermes, OpenClaw, or any OpenAI-compatible client OPENAI_BASE_URL : https://your-project.vercel.app/v1 OPENAI_API_KEY : any-key (gateway-managed load balancer) # Recommended default models: Model (Primary) : gemini/gemini-3.5-flash-lite → 250K TPM, agent-capable Model (Cheapest): gemini/gemini-3.1-flash-lite → 250K TPM, high throughput
